        const queryRes = await fetch(queryUrl, { method: 'POST' });

        const audioQuery = await queryRes.json();

        const synthUrl = `${VOICEVOX_URL}/synthesis?speaker=${speakerId}`;
        const synthRes = await fetch(synthUrl, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(audioQuery)
        });
        if (!synthRes.ok) throw new Error('synthesis 失敗');

        const audioBlob = await synthRes.blob();
        const audioUrl = URL.createObjectURL(audioBlob);
        const audio = new Audio(audioUrl);

    function initSpeechRecognition() {
      const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
      if (!SpeechRecognition) {
        speechRecognizeStatusInfo.textContent = '音声認識状態: 未対応ブラウザです';
        return;
      }

      recognition = new SpeechRecognition();
      recognition.lang = 'ja-JP';
      recognition.continuous = true;
      recognition.interimResults = false;

      recognition.onstart = () => {
        isListening = true;
        speechRecognizeStatusInfo.textContent = '音声認識状態: 常時マイク待機中...';
      };

      recognition.onend = () => {
        isListening = false;
        if (speechAssistantToggle.checked) {
          try { recognition.start(); } catch(e) {}
        } else {
          speechRecognizeStatusInfo.textContent = '音声認識状態: 停止中';
        }
      };

      recognition.onerror = (e) => {
        console.warn('音声認識エラー:', e);
      };

      recognition.onresult = (e) => {
        if (!speechAssistantToggle.checked || isSpeechCooldown) return;
        const lastIndex = e.results.length - 1;
        const text = e.results[lastIndex][0].transcript.trim();
        console.log('認識テキスト:', text);

        handleSpeechQuery(text);
      };

      if (speechAssistantToggle.checked) {
        try { recognition.start(); } catch(e) {}
      }
    }

    function handleSpeechQuery(inputText) {
      if (isSpeechCooldown) return;

      const callKeywords = callKeywordsInput.value.split(',').map(k => k.trim()).filter(k => k);
      const isCalled = callKeywords.some(kw => inputText.includes(kw));

      if (isCalled || aiQaBox.classList.contains('is-active')) {
        aiQaBox.classList.add('is-active');
        aiSpeechStatus.textContent = `🎙️ 認識: 「${inputText}」`;

        let matched = knowledgeBase.find(item => 
          item.keywords && item.keywords.some(kw => inputText.includes(kw))
        );

        if (!matched) {
          matched = defaultFallback;
        }

        aiResponseText.textContent = matched.response;
        renderSuggestions(matched.suggestions || []);
        speakInputText(matched.response);

        const cooldownSec = Math.max(0, parseInt(speechCooldownInput.value) || 0);
        if (cooldownSec > 0) {
          isSpeechCooldown = true;
          speechRecognizeStatusInfo.textContent = `音声認識状態: 応答クールダウン中 (${cooldownSec}秒)`;
          setTimeout(() => {
            isSpeechCooldown = false;
            if (speechAssistantToggle.checked) {
              speechRecognizeStatusInfo.textContent = '音声認識状態: 常時マイク待機中...';
            }
          }, cooldownSec * 1000);
        }
      }
    }

    function playWinSE() {
      try {
        const AudioCtx = window.AudioContext || window.webkitAudioContext;
        if (!AudioCtx) return;
        const ctx = new AudioCtx();
        const notes = [523.25, 659.25, 783.99, 1046.50];
        notes.forEach((freq, idx) => {
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.type = 'triangle';
          osc.frequency.value = freq;
          
          const startTime = ctx.currentTime + (idx * 0.08);
          gain.gain.setValueAtTime(0.3, startTime);
          gain.gain.exponentialRampToValueAtTime(0.001, startTime + 0.3);
          
          osc.connect(gain);
          gain.connect(ctx.destination);
          osc.start(startTime);
          osc.stop(startTime + 0.3);
        });
      } catch (e) {
        console.warn('確定SE再生エラー:', e);
      }
    }

    function loadAndProcessBgImage(src) {
      currentBgImageSrc = src;
      const img = new Image();
      img.crossOrigin = "anonymous";
      img.onload = () => {
        buildingBgCanvas.width = img.width;
        buildingBgCanvas.height = img.height;
        const ctx = buildingBgCanvas.getContext('2d');
        ctx.drawImage(img, 0, 0);

        if (chromaKeyToggle.checked) {
          const imgData = ctx.getImageData(0, 0, img.width, img.height);
          const data = imgData.data;
          for (let i = 0; i < data.length; i += 4) {
            const r = data[i];
            const g = data[i + 1];
            const b = data[i + 2];
            if (g > 80 && g > r * 1.3 && g > b * 1.3) {
              data[i + 3] = 0;
            }
          }
          ctx.putImageData(imgData, 0, 0);
        }
      };
      img.src = src;
    }

    function applyPartTransforms() {
      Object.keys(partConfigs).forEach(key => {
        const conf = partConfigs[key];
        let elements = [];
        if (key === 'eye') {
          elements = [document.getElementById('part-eye'), document.getElementById('part-eye-close')];
        } else if (key === 'mouth') {
          elements = [document.getElementById('part-mouth-open'), document.getElementById('part-mouth-close')];
        } else {
          const el = document.getElementById(`part-${key}`);
          if (el) elements.push(el);
        }

        elements.forEach(el => {
          if (el) {
            const scaleFactor = conf.scale / 100;
            el.style.transform = `translate(${conf.x}px, ${conf.y}px) scale(${scaleFactor}) rotate(${conf.rotate}deg)`;
          }
        });
      });
    }

        } else {
          el.classList.remove('part-is-locked');
        }
      });

      document.querySelectorAll('.part-lock-chk').forEach(chk => {
        const key = chk.getAttribute('data-key');
        chk.checked = !!lockedParts[key];

        const e2 = document.getElementById('part-eye-close');

    let voiceList = JSON.parse(localStorage.getItem('momonga_voice_list')) || ['talk_01.wav', 'talk_02.wav'];

    let voiceTimer = null;
    let currentVoiceIndex = 0;

    function saveVoiceList() {
      localStorage.setItem('momonga_voice_list', JSON.stringify(voiceList));
    }

    function renderVoiceListUI() {
      voiceListContainer.innerHTML = '';
      voiceList.forEach((file, idx) => {
        const row = document.createElement('div');
        row.className = 'sound-item-row';

        const radio = document.createElement('input');
        radio.type = 'radio';
        radio.name = 'selectedVoiceRadio';
        radio.value = idx;
        radio.checked = (idx === currentVoiceIndex);
        radio.addEventListener('change', () => {
          currentVoiceIndex = idx;
        });

        const input = document.createElement('input');
        input.type = 'text';
        input.value = file;
        input.addEventListener('input', (e) => {
          voiceList[idx] = e.target.value.trim();
          saveVoiceList();
        });

        const delBtn = document.createElement('button');
        delBtn.textContent = '✕';
        delBtn.style.background = '#ff0055';
        delBtn.style.marginTop = '0';
        delBtn.style.padding = '4px 8px';
        delBtn.addEventListener('click', () => {
          voiceList.splice(idx, 1);
          if (currentVoiceIndex >= voiceList.length) currentVoiceIndex = Math.max(0, voiceList.length - 1);
          saveVoiceList();
          renderVoiceListUI();
        });

        row.appendChild(radio);
        row.appendChild(input);
        row.appendChild(delBtn);
        voiceListContainer.appendChild(row);
      });
    }

    addVoiceBtn.addEventListener('click', () => {
      voiceList.push(`talk_0${voiceList.length + 1}.wav`);
      saveVoiceList();
      renderVoiceListUI();
    });

    renderVoiceListUI();

    function playVoiceFile(filePath) {
      if (!filePath) return;
      const audio = new Audio(filePath);
      audio.volume = parseFloat(volumeRange.value) || 0.5;

      audio.addEventListener('loadedmetadata', () => {
        const durationMs = Math.max(1000, Math.ceil(audio.duration * 1000));
        startTalking(durationMs);
        momongaContainer.classList.add('bounce');
        setTimeout(() => momongaContainer.classList.remove('bounce'), 800);
      });

      audio.play().catch(err => console.warn('ボイス再生失敗:', err));
    }

      if (voiceList.length > 0 && voiceList[currentVoiceIndex]) {
        playVoiceFile(voiceList[currentVoiceIndex]);
      } else {
        alert('ボイスが登録されていません。');
      }
    });

    function scheduleVoice() {
      clearTimeout(voiceTimer);
      if (!isVoiceEnabled || voiceList.length === 0) {
        nextSoundInfo.textContent = '次回再生まで: 停止中';
        return;
      }

      const minMin = Math.max(1, parseInt(minIntervalInput.value) || 1);
      const maxMin = Math.max(minMin, parseInt(maxIntervalInput.value) || 3);
      const randMin = Math.floor(Math.random() * (maxMin - minMin + 1)) + minMin;
      const delayMs = randMin * 60 * 1000;

      nextSoundInfo.textContent = `次回再生: 約${randMin}分後`;

      voiceTimer = setTimeout(() => {
        const mode = voicePlayMode.value;
        let fileToPlay = '';

        if (mode === 'single') {
          fileToPlay = voiceList[currentVoiceIndex] || voiceList[0];
        } else if (mode === 'random') {
          const rIdx = Math.floor(Math.random() * voiceList.length);
          fileToPlay = voiceList[rIdx];
        } else if (mode === 'sequence') {
          fileToPlay = voiceList[currentVoiceIndex];
          currentVoiceIndex = (currentVoiceIndex + 1) % voiceList.length;
          renderVoiceListUI();
        }

        playVoiceFile(fileToPlay);
        scheduleVoice();
      }, delayMs);
    }

        soundToggleBtn.style.color = '#fff';
